startEvent("Patient Arrival in ED")

userTask("Triage Nurse Assesses Patient")
  timerBoundary("PT15M"):  // ED protocol: Triage within 15 minutes

if (acuity == 1 AND bed_available == true):
    serviceTask("Admit to Monitored Bed")
else if (acuity <= 5 AND bed_available == false):
    userTask("Patient Waits in ED Area")
else:
    serviceTask("Admit to Standard ED Bay")  // Default path

// --- ASSESSMENT PHASE ---
userTask("Physician Performs Assessment")
  timerBoundary("PT4H"):
    sendTask("Send: Escalation to Department Head")
  errorBoundary():
    userTask("Senior Physician Takes Over Assessment")

serviceTask("Order Diagnostic Tests")

// --- DISPOSITION DECISION ---
if (admission_required == true):
    subProcess("Admission Workflow"):
        startEvent("Admission Initiated")

        serviceTask("Check Bed Availability")
          errorBoundary():
            userTask("Manual Bed Assignment by Admissions")

        if (icu_bed_available == true AND acuity == 1):
            serviceTask("Assign ICU Bed")
        else if (general_bed_available == true AND acuity <= 3):
            serviceTask("Assign General Ward Bed")
        else:
            serviceTask("Place in Waiting Queue")
            sendTask("Send: Notification to Nursing Manager")
            endEvent("Waiting Queue Placement")

        serviceTask("Transfer Patient to Assigned Bed")
        serviceTask("Initiate Admission Protocols")

        subProcess("Medication Reconciliation"):
            startEvent("Med Rec Started")
            serviceTask("Verify Allergies")
            serviceTask("Pharmacy Reviews Medications")
              timerBoundary("PT4H"):
                sendTask("Send: Urgent Pharmacy Alert")

            if (interaction_found == true):
                userTask("Prescribing Physician Adjusts Medication Regimen")
                endEvent("Regimen Adjusted")
            else:
                endEvent("No Interactions")
        endSubProcess

        serviceTask("Track Insurance Authorization")
        if (authorization_approved == false):
            userTask("Billing Team Contacts Patient for Approval")

        endEvent("Admission Complete")
    endSubProcess
else:
    userTask("Physician Documents Discharge Instructions")
    endEvent("Discharge Complete")

// --- CONTINUOUS MONITORING ---
serviceTask("Monitor Physician Orders")
  if (critical_lab_value_detected == true):
    sendTask("Send: Critical Lab Alert to Physician")
    userTask("Physician Reviews Alert")
