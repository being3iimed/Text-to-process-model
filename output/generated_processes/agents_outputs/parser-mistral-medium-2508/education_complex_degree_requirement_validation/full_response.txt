```pseudocode
startEvent("Graduation Check Requested")

// --- SUBPROCESS: Validate Requirements ---
subProcess("Validate Graduation Requirements"):
    startEvent("Validation Start")

    // Parallel checks (AND gateway)
    AND:
        // Academic checks
        scriptTask("Computation: Sum total credits")
        scriptTask("Computation: Verify ≥120 credits")
        scriptTask("Computation: Verify cumulative GPA ≥ 2.0")
        scriptTask("Computation: Verify major GPA ≥ [major_threshold]")

        // Administrative checks
        serviceTask("Check disciplinary holds")
        serviceTask("Check library fees")
        if (student.is_international):
            serviceTask("Verify visa status")
        END_IF

        // Course requirements
        serviceTask("Verify all major courses completed")
        serviceTask("Verify all general education requirements met")
    END_AND

    // Converge parallel checks
    exclusiveGateway("All Requirements Met?"):
        if (all_requirements_met):
            endEvent("Validation Complete - Approved")
        else:
            serviceTask("Generate Detailed Deficiency Report"):
                timerBoundary("P30D"):  // 30-day deadline to resolve
                    sendTask("Send: Urgent - Deadline Missed Notification")
            endEvent("Validation Complete - Deficiencies Found")
endSubProcess

// --- Main Flow ---
exclusiveGateway("Validation Result"):
    if (approved):
        // --- SUBPROCESS: Fulfill Graduation ---
        subProcess("Fulfill Graduation"):
            startEvent("Fulfillment Start")

            AND:
                serviceTask("Prepare Diploma")
                serviceTask("Order Cap and Gown")
                serviceTask("Assign Ceremony Slot")
            END_AND

            // Ceremony day
            userTask("Verify Attendance at Ceremony")  // Manual check-in
            serviceTask("Confer Degree")
            serviceTask("Issue Transcript")

            endEvent("Fulfillment Complete")
        endSubProcess
    else:
        sendTask("Send: Deficiency Report to Student")
        endEvent("Graduation Denied")
    END_IF

endEvent("Process Complete")
```
