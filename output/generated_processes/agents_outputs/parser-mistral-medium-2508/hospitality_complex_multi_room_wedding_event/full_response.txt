startEvent("Wedding Contract Signed")

// Initial Planning Phase
userTask("Sign Contract")
userTask("Confirm Ceremony Time")
AND:
    userTask("Finalize Guest Count")
    userTask("Assign Guest Rooms")
    userTask("Select Menu Items")
    if (has_dietary_restrictions):
        userTask("Specify Dietary Restrictions")
    else:
        serviceTask("Note: no restrictions")
    END_IF
    userTask("Choose Beverage Selections")
    userTask("Arrange Music/DJ")
END_AND

// Scheduled Milestones
timerStartEvent("3 Months Before Event"):
    userTask("Coordinator Creates Master Timeline")
    serviceTask("Distribute to Departments")

timerStartEvent("2 Weeks Before Event"):
    userTask("Final Headcount Confirmation"):
        timerBoundary("PT4H"):
            sendTask("Send: late fee notification")
    userTask("Room Block Finalization")
    userTask("Dietary Restrictions List Finalized")
    serviceTask("Reserve AV Equipment")

timerStartEvent("1 Week Before Event"):
    userTask("Facility Walkthrough"):
        timerBoundary("PT24H"):
            sendTask("Send: walkthrough overdue alert")
    userTask("Finalize Seating Chart")
    serviceTask("Prepare Name Placards")
    scriptTask("Computation: calculate alcohol quantities by guest count")

// Day-of Execution
timerStartEvent("Day of Event - 8:00 AM"):
    serviceTask("Morning Setup"):
        AND:
            serviceTask("Arrange Ceremony Room")
            serviceTask("Arrange Reception Tables")
        END_AND
    serviceTask("AV Equipment Test")

    // Ceremony Workflow
    intermediateEvent(timerBoundary("PT2H")) // Buffer before ceremony
    userTask("Guest Check-in Processing")
    userTask("Conduct Ceremony")

    // Reception Workflow
    subProcess("Catering Preparation"):
        startEvent("Kitchen Prep Begins")
        serviceTask("Finalize Cooking Schedule")
        serviceTask("Prep Cold Items")
        serviceTask("Coordinate Plating")
        serviceTask("Set Up Buffet Stations")
        serviceTask("Execute Service Timeline")
        endEvent("Catering Service Complete")
    endSubProcess

    subProcess("Bar Service Operation"):
        startEvent("Bar Setup Begins")
        serviceTask("Inventory Verification")
        serviceTask("Staff Briefing")
        serviceTask("Open Bar Service")
        serviceTask("Monitor Consumption")
        serviceTask("Close Bar Service")
        endEvent("Bar Service Complete")
    endSubProcess

    serviceTask("Dancing/Entertainment")
    intermediateEvent(timerBoundary("PT1H")) // Buffer before cleanup

// Post-Event
serviceTask("Post-Event Cleanup"):
    AND:
        serviceTask("Remove Furniture")
        serviceTask("Reset Facility")
    END_AND
userTask("Guest Checkout with Luggage Storage")
scriptTask("Computation: compare actual vs estimated charges")
serviceTask("Generate Final Bill")
endEvent("Final Billing Reconciled")

// Parallel Guest Room Process
AND:
    receiveTask("Receive: guest room requests"):
        timerBoundary("PT24H"):
            sendTask("Send: room assignment confirmation")
    serviceTask("Process Room Upgrades")
    userTask("Handle Special Requests")
    endEvent("Guest Room Process Complete")
END_AND
</pseudocode>
