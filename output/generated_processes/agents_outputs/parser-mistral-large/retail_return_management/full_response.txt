```pseudocode
startEvent("Return Process Initiated")

// Customer initiates return
userTask("Initiate Return Request")
  name: "Customer submits order number and reason"
  inputs:
    - order_number
    - return_reason

// System validation
serviceTask("Validate Return Eligibility")
  name: "System checks 30-day window and item condition"
  inputs:
    - order_number
    - return_reason
  outputs:
    - eligible (boolean)
    - validation_message

// Eligibility decision
if (eligible == true):
    sendTask("Send: return shipping label")
      name: "System generates and emails return label"
      inputs:
        - order_number
      outputs:
        - shipping_label_sent (boolean)

    userTask("Ship Item Back")
      name: "Customer ships item using provided label"
      inputs:
        - shipping_label_sent

    userTask("Verify Item Condition")
      name: "Warehouse staff inspects returned item"
      inputs:
        - returned_item
      outputs:
        - condition (enum: acceptable/damaged)
        - verification_notes

    // Condition decision
    if (condition == acceptable):
        serviceTask("Process Refund")
          name: "System issues refund to original payment method"
          inputs:
            - order_number
            - refund_amount
          outputs:
            - refund_processed (boolean)

        sendTask("Send: refund confirmation email")
          name: "System sends confirmation to customer"
          inputs:
            - order_number
            - refund_amount
          outputs:
            - confirmation_sent (boolean)

        endEvent("Return Process Complete")

    else: // condition == damaged
        sendTask("Send: return denial notification")
          name: "System notifies customer of damaged item"
          inputs:
            - order_number
            - verification_notes
          outputs:
            - denial_sent (boolean)

        endEvent("Return Denied")

else: // eligible == false
    endEvent("Return Denied")
      name: "Ineligible return request terminated"
```
