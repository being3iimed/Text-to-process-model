if (prior_auth_required == true):

        sendTask("Send: pickup notification to patient")
        userTask("Patient Signs Receipt")
        serviceTask("Dispense Medication")
        endEvent("Prescription Dispensed")

        sendTask("Send: denial notification to patient and doctor")

        if (patient_action == 'appeal'):
            endEvent("Appeal Process Initiated")
        else: // doctor_action == 'prescribe_alternative'
            endEvent("Alternative Prescribed")

else: // prior_auth_required == false (default)
    serviceTask("Fill Prescription")
    sendTask("Send: pickup notification to patient")
    userTask("Patient Signs Receipt")
    serviceTask("Dispense Medication")
    endEvent("Prescription Dispensed")
