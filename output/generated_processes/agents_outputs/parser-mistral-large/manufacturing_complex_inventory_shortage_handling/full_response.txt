### **2. BPMN Pseudocode**
```pseudocode
startEvent("Component X Shortage Detected")

// --- Check Next Delivery ---
serviceTask("Query Supplier Delivery Schedule")
if (next_delivery_eta <= "PT24H"):
    serviceTask("Log: Minor Production Delay")
    serviceTask("Adjust Production Schedule")
    endEvent("Production Resumes with Original Components")
else:
    // --- Check Alternative Component ---
    serviceTask("Query Alternative Components Database")
    if (alternative_available):
        sendTask("Send: Use Alternative Component to Production")
        serviceTask("Log: Alternative Component Used")
        serviceTask("Adjust Production Parameters if Needed")
        endEvent("Production Resumes with Alternative")
    else:
        // --- Expedite Shipment Subprocess ---
        subProcess("Expedite Shipment"):
            startEvent("Expedite Request Initiated")
            userTask("Production Manager Approves Expedite Premium Cost")
            serviceTask("Send Expedite Request to Supplier")
            if (supplier_agrees):
                sendTask("Send: Confirmation to Production Line")
                sendTask("Send: Delay Notice to Dependent Lines")
                endEvent("Expedite Confirmed")
            else:
                endEvent("Expedite Rejected")
        endSubProcess

        if (expedite_confirmed):
            serviceTask("Update Delivery ETA")
            serviceTask("Log: Expedite Cost Variance")
            endEvent("Production Resumes with Expedited Components")
        else:
            // --- Backup Supplier Subprocess ---
            subProcess("Procure Backup Supplier"):
                startEvent("Escalation to Procurement")
                userTask("Procurement Manager Selects Backup Supplier")
                serviceTask("Verify Backup Supplier Stock")
                serviceTask("Negotiate Same-Day Delivery")
                if (backup_confirmed):
                    serviceTask("Create Purchase Order for Backup Supplier")
                    serviceTask("Log: Backup Supplier Cost Variance")
                    endEvent("Backup Supplier Secured")
                else:
                    endEvent("No Backup Available")
            endSubProcess

            if (backup_secured):
                serviceTask("Update Production Schedule")
                serviceTask("Update Supplier Performance Metrics")
                endEvent("Production Resumes with Backup Components")
            else:
                userTask("Manual Resolution Required")
                endEvent("Process Terminated: No Resolution")
            endif
        endif
    endif
endif
```
