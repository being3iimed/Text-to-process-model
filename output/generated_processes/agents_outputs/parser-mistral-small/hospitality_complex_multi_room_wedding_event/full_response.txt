startEvent("Wedding Planning Begins")

subProcess("Wedding Planning"):
    startEvent("Planning Start")
    userTask("Select Hotel")
    serviceTask("Enter Contract")
    userTask("Confirm Details with Coordinator")
    serviceTask("Create Master Timeline")
    userTask("Confirm Final Details"):
        timerBoundary("2 weeks pre-event"):
            sendTask("Send: reminder to confirm details")

AND:
    serviceTask("Finalize Seating Chart")

userTask("Morning Setup")

    END_AND

serviceTask("Immediate Cleanup")
