startEvent("Prescription Process")
serviceTask("Send prescription to pharmacy")
receiveTask("Receive: prescription")
userTask("Check insurance eligibility")
if (prior_authorization_required):
    userTask("Contact insurance company")
    if (insurance_approved):
        serviceTask("Fill prescription")
        sendTask("Send: notification to patient")
        userTask("Sign receipt")
        endEvent("Process Complete")
    else:
        sendTask("Send: denial notification")
        OR:
            userTask("Process appeal")
            serviceTask("Fill prescription")
            sendTask("Send: notification to patient")
            userTask("Sign receipt")
            endEvent("Process Complete")
            userTask("Prescribe alternative")
            endEvent("Denial Processed")
        END_OR
else:
    serviceTask("Fill prescription")
    sendTask("Send: notification to patient")
    userTask("Sign receipt")
    endEvent("Process Complete")
