userTask("Check Course Details")

    serviceTask("Verify Prerequisites")
    serviceTask("Check Seat Availability")
    userTask("Enroll in Course")
    serviceTask("Confirm Enrollment")
    serviceTask("Generate Course Schedule")
    serviceTask("Update Course Roster")
    endEvent("Enrollment Complete")
else:
    endEvent("Enrollment Not Possible")
