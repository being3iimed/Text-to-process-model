    serviceTask("Submit Injury Report")
        timerBoundary("24 hours"):
            sendTask("Send: regulatory violation notice")
        errorBoundary():
            userTask("Resubmit Injury Report")

    userTask("Assign Claims Adjuster")
    sendTask("Send: Initial Contact")
    userTask("Collect Injury Details")

    if (injury_work_related):

        endEvent("Assessment Failed")
