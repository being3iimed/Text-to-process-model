receiveTask("Receive: Referred Party Action (purchase/signup)")

    serviceTask("Calculate Reward Amount")
    AND:

    userTask("Referrer Redeems Reward")
    endEvent("Reward Redeemed")
else:
    sendTask("Send: Notification to Referrer (eligibility not met)")
    endEvent("Reward Not Earned")

    endEvent("Reward Expired")
