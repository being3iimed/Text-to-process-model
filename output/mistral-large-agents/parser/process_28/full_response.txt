startEvent("Internship Application Started")

// --- Phase 1: Preferences & Offers ---
userTask("Submit Internship Preferences")
serviceTask("Generate Internship Offers")

XOR:
  if (offer_received == true):
    userTask("Review Offer")
    XOR:
      if (user_accepts):
        userTask("Accept Offer")
        serviceTask("Invalidate Other Offers")
        AND:  // Parallel tracks post-acceptance
          subProcess("Status Update Cycle"):
            startEvent("Status Updates Begin")
            loop (3 times):
              userTask("Write Weekly Status Update")
              serviceTask("Generate Company Status Update")
              serviceTask("Store Updates")
            endEvent("All Updates Complete")
          endSubProcess

          intermediateEvent(timerBoundary("30 days"))  // SLA for internship duration
          endEvent("Internship Auto-Completed")
        END_AND
      else:  // user_rejects (default)
        serviceTask("Log Rejection")
        intermediateEvent(timerBoundary("7 days"))  // Wait for next offer
        loop_back_to: serviceTask("Generate Internship Offers")
  else:  // no_offers_yet (default)
    intermediateEvent(timerBoundary("7 days"))
    loop_back_to: serviceTask("Generate Internship Offers")

// --- Phase 2: Recommendations ---
userTask("Select Friends to Recommend")
sendTask("Send: Recommendation Tweets")  // Parallel tweets via AND split (implied by "separate tweets")

endEvent("Recommendations Sent")
endEvent("Internship Completed")  // Convergence from status updates and timer
