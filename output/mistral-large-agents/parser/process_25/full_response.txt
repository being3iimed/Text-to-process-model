timerStartEvent("Weekly on Monday at 09:00"):
    subProcess("Application Reporting"):

        receiveTask("Receive Job Offer")

        subProcess("Probation Phase"):
            startEvent("Probation Begins")

            userTask("Confirm Probation Completion")
            receiveTask("Receive Probation Outcome")

            if (probation_outcome == "permanent"):
                serviceTask("Mark Job as Permanent")
                userTask("Review Company Rating")

                if (company_rating > 'C'):
                    endEvent("Process Complete - Permanent Job")
                else:
                    // Continue receiving offers without reporting
                    serviceTask("Update Preferences: No Reporting")
                    loopBackTo: "Generate Potential Offers"
            else: // probation_outcome == "terminated"
                endEvent("Process Complete - Probation Failed")
            endif

            // Review visibility timer (parallel to main flow)
            intermediateEvent(timerBoundary("1 year"))
            sendTask("Send: Review Visibility Notification")

            endEvent("Probation Phase Complete")
        endSubProcess
    else: // application_rating < threshold
        loopBackTo: "Generate Potential Offers"
    endif

endEvent("Process Complete - Voluntary Exit")
