<bpmn_pseudocode>

startEvent("Claim Registered")
    userTask("Examine Claim")
    userTask("Write Settlement Recommendation")

    userTask("Review Recommendation")
    if (recommendation == "OK"):
        endEvent("Recommendation Approved")
    else:  // recommendation == "Not OK" (default)
        userTask("Revise Recommendation")
        // Loop back to senior review
        userTask("Review Recommendation")
        if (recommendation == "OK"):
            endEvent("Recommendation Approved")
        else:
            // Prevent infinite loop - force decision after 2nd review
            userTask("Final Decision by Manager")
            endEvent("Recommendation Approved")
