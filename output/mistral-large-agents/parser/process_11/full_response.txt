<bpmn_pseudocode>

## BPMN Pseudocode

```pseudocode
// MAIN PROCESS: Subscription Lifecycle
startEvent("Customer Initiates Signup")

// Initial Setup
userTask("Enter Subscription Information")
serviceTask("Generate Account")
serviceTask("Assign Access")
serviceTask("Process Initial Payment")
sendTask("Send: Welcome Email and Access Instructions")
endEvent("Subscription Active")

// Recurring Billing (Subprocess)
subProcess("Monthly Billing Cycle"):
    timerStartEvent("Monthly Billing Trigger")
        "$type": "bpmn:TimerEventDefinition",
        "timeCycle": "RRULE:FREQ=MONTHLY;BYMONTHDAY=1;BYHOUR=0;BYMINUTE=0"

    sendTask("Send: Renewal Notification (7 days prior)")
    serviceTask("Generate Invoice")
    serviceTask("Process Billing Payment")

    if (payment_successful):
        sendTask("Send: Payment Receipt")
        serviceTask("Send Product Updates")
        serviceTask("Send Enhancement Notifications")
        endEvent("Billing Cycle Complete")
    else:
        sendTask("Send: Payment Failed Notification")
        userTask("Customer Updates Payment Method")
        serviceTask("Retry Billing Payment")
        endEvent("Billing Cycle Complete")
endSubProcess

// Cancellation Handling (Subprocess)
subProcess("Cancellation Processing"):
    messageStartEvent("Cancellation Request Received")

    userTask("Support Team Reviews Cancellation")
    if (cancellation_approved):
        serviceTask("Calculate Final Balance")
        if (refund_due > 0):
            serviceTask("Process Refund")
            sendTask("Send: Refund Confirmation")
        else:
            serviceTask("Apply Final Charges")
            sendTask("Send: Final Invoice")
        endIf

        serviceTask("Deactivate Subscription")
        sendTask("Send: Cancellation Confirmation")
        endEvent("Subscription Cancelled")
    else:
        sendTask("Send: Cancellation Rejection Notice")
        endEvent("Cancellation Denied")
    endIf
endSubProcess
```
