// MAIN PROCESS: Workers Compensation Claim
startEvent("Workplace Injury Occurs")

// Initial Reporting Phase
userTask("Report Injury to Employer")
serviceTask("Employer Submits Injury Report to Insurer"):
    timerBoundary("24 hours"):
        sendTask("Send: SLA Violation Notice to Employer")

// Claim Assignment & Initial Assessment
userTask("Adjuster Contacts Injured Worker")
userTask("Adjuster Determines Work-Related Injury")
if (injury_is_work_related):
    userTask("Adjuster Determines Applicable Benefits")
    if (medical_coverage_applies OR disability_applies):
        serviceTask("Verify Worker's Wage and Employment History")

        // Medical Treatment Management Subprocess
        subProcess("Medical Treatment Management"):
            startEvent("Treatment Management Begins")

            AND:
                userTask("Authorize Medical Treatment")
                userTask("Review Medical Reports")
            END_AND

            if (dispute_about_injury_severity):
                userTask("Order Independent Medical Examination (IME)")
                userTask("IME Physician Provides Second Opinion")
            endIf

            endEvent("Treatment Management Complete")
        endSubProcess

        // Return-to-Work Coordination Subprocess
        subProcess("Return-to-Work Coordination"):
            startEvent("RTW Coordination Begins")

            userTask("Monitor Return-to-Work Status")
            if (worker_can_return):
                userTask("Coordinate Job Placement with Employer")
            endIf

            serviceTask("Submit Status Report to State Board")
            endEvent("RTW Coordination Complete")
        endSubProcess

        // Final Assessment Phase
        userTask("Calculate Permanent Disability Award")
    else:
        sendTask("Send: Benefits Denial Notice")
    endIf
else:
    sendTask("Send: Non-Covered Injury Notice")
endIf

serviceTask("Close Claim File")
endEvent("Claim File Closed")
