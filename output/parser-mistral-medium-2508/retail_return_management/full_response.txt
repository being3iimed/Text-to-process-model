<bpmn_pseudocode>

startEvent("Return Process Initiated")

// Initial customer action
userTask("Customer Initiates Return")
  inputs: order_number, return_reason

// System validation
serviceTask("Validate Return Eligibility")
  logic: (order_date >= now-30days) AND (item_condition in ["new","used_like_new"])

// Eligibility decision
if (eligible):
    sendTask("Send: return shipping label")
    userTask("Customer Ships Item")
    userTask("Warehouse Receives Package")
    userTask("Verify Item Condition")

    // Condition verification
    if (condition == "acceptable"):
        serviceTask("Process Refund")
        sendTask("Send: refund confirmation email")
        endEvent("Return Process Complete")
    else: // condition == "damaged"
        sendTask("Send: return denial notice")
        endEvent("Return Denied")
else: // ineligible
    sendTask("Send: return denial notice")
    endEvent("Return Denied")
