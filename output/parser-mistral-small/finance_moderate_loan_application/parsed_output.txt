    endEvent("Process Complete")

    if (additional_documents_needed):
        userTask("Request Additional Documentation")
        receiveTask("Receive: Additional Documentation")
    END_IF

    if (approved):
        sendTask("Send: Loan Approval Terms")

        endEvent("Process Complete")
