### SUMMARY OF ELEMENTS
--------------------------------------------------------------------------------
startEvent("Graduation Check Requested")

subProcess("Verify Academic Requirements"):
    startEvent("Academic Verification Start")
    serviceTask("Verify Course Completions")
    serviceTask("Verify Credit Count")
    serviceTask("Verify Cumulative GPA")
    serviceTask("Verify Major GPA")
    endEvent("Academic Verification Complete")
endSubProcess

subProcess("Verify Administrative Requirements"):
    startEvent("Administrative Verification Start")
    serviceTask("Verify Disciplinary Holds")
    serviceTask("Verify Library Fees")
    serviceTask("Verify Visa Status")
    endEvent("Administrative Verification Complete")
endSubProcess

if (all requirements met):
    subProcess("Graduation Preparation"):
        startEvent("Graduation Preparation Start")
        AND:
            serviceTask("Prepare Diploma")
            serviceTask("Place Cap-and-Gown Order")
            serviceTask("Assign Graduation Slot")
        END_AND
        endEvent("Graduation Preparation Complete")
    endSubProcess

    serviceTask("Verify Attendance")
    serviceTask("Confer Degree")
    serviceTask("Issue Transcript")
else:
    serviceTask("Generate Incomplete Report")

endEvent("Graduation Process Complete")
