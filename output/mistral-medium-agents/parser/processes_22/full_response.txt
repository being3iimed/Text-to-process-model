startEvent("Subrogation case received")

// Initial assessment
userTask("Check subrogation possibility")

if (subrogation_possible):
    // Payment request path
    serviceTask("Send payment request to insurant")

    // Business-required reminder
    intermediateEvent(timerBoundary("7 days"))
    sendTask("Send: reminder to insurant")

    // Wait for response with SLA
    receiveTask("Receive: payment or disagreement"):
        timerBoundary("30 days"):
            sendTask("Send: case to collection agency")
            endEvent("Case forwarded to collection")

    if (payment_received):
        serviceTask("Make financial booking")
        endEvent("Case closed - payment booked")

    else if (disagreement_received):
        // Reasoning evaluation
        userTask("Check disagreement reasoning")

        if (insurant_correct):
            endEvent("Case closed - insurant correct")
        else:
            sendTask("Send: case to collection agency")
            endEvent("Case forwarded to collection")

    else: // timer_expired (default)
        // Handled by boundary event above
        [unreachable - boundary takes precedence]

else: // subrogation_not_possible (default)
    endEvent("Case closed - no subrogation")
