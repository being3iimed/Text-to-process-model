// --- Auditor Review ---
userTask("Auditors Review Processes and Identify Gaps")

// --- Gap Resolution or Proceed ---
if (gaps_identified):
    userTask("Implement Corrections/Improvements")
else:
    // Default: No gaps, proceed to final audit
END_IF

// --- Final Audit ---
userTask("Final Audit by External Auditors")

// --- Certification Decision ---
if (final_audit_passed):
    serviceTask("Issue Compliance Certification")
    endEvent("Certification Awarded")
else:
    endEvent("Audit Failed - Process Terminated")
END_IF
